cmake_minimum_required(VERSION 3.25)
project(SwitchCore VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(NEXO "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nexo")
set(SRC "${NEXO}/src")
file(GLOB_RECURSE ENGINE CONFIGURE_DEPENDS "${SRC}/core/*.cpp")
file(GLOB MAXWELL CONFIGURE_DEPENDS "${SRC}/video_core/*.cpp")
add_library(switchcore STATIC ${ENGINE} ${MAXWELL})
target_include_directories(switchcore PUBLIC "${SRC}" "${SRC}/core" "${SRC}/core/arm64" "${SRC}/core/memory" "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(Threads REQUIRED)
target_link_libraries(switchcore PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(switchcore PRIVATE ws2_32 crypt32)
endif()
if(MSVC)
  target_compile_options(switchcore PRIVATE /utf-8 /bigobj)
  set_source_files_properties("${SRC}/core/arm64/fp_ops.cpp" PROPERTIES COMPILE_OPTIONS /fp:strict)
else()
  set_source_files_properties("${SRC}/core/arm64/fp_ops.cpp" PROPERTIES COMPILE_OPTIONS -frounding-math)
endif()
include(CTest)
if(BUILD_TESTING)
  file(GLOB ENGINE_TESTS CONFIGURE_DEPENDS "${NEXO}/tests/*.cpp")
  list(FILTER ENGINE_TESTS EXCLUDE REGEX "vulkan_tests[.]cpp$")
  add_executable(engine-tests ${ENGINE_TESTS})
  target_link_libraries(engine-tests PRIVATE switchcore)
  target_compile_definitions(engine-tests PRIVATE NEXO2_TEST_DATA_DIR="${NEXO}/tests/generated")
  add_test(NAME engine COMMAND engine-tests --interp)
  set_tests_properties(engine PROPERTIES TIMEOUT 240)
endif()

target_sources(switchcore PRIVATE core/runtime/SwitchCore.cpp)
if(BUILD_TESTING)
  add_executable(switchcore-probe tools/probe.cpp)
  target_link_libraries(switchcore-probe PRIVATE switchcore)
  add_test(NAME interactive-rally COMMAND switchcore-probe "${CMAKE_CURRENT_SOURCE_DIR}/apps/ios/Resources/rally.nro" "${CMAKE_CURRENT_BINARY_DIR}/rally-title" rally "${CMAKE_CURRENT_BINARY_DIR}/rally.ppm")
endif()
