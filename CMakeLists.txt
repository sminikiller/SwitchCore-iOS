cmake_minimum_required(VERSION 3.25)
project(SwitchCore VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(NEXO "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nexo")
set(SRC "${NEXO}/src")
file(GLOB_RECURSE ENGINE CONFIGURE_DEPENDS "${SRC}/core/*.cpp")
file(GLOB MAXWELL CONFIGURE_DEPENDS "${SRC}/video_core/*.cpp")
add_library(switchcore STATIC ${ENGINE} ${MAXWELL})
target_include_directories(switchcore PUBLIC "${SRC}" "${SRC}/core" "${SRC}/core/arm64" "${SRC}/core/memory" "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(Threads REQUIRED)
target_link_libraries(switchcore PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(switchcore PRIVATE ws2_32 crypt32)
endif()
if(MSVC)
  target_compile_options(switchcore PRIVATE /utf-8 /bigobj)
  set_source_files_properties("${SRC}/core/arm64/fp_ops.cpp" PROPERTIES COMPILE_OPTIONS /fp:strict)
else()
  set_source_files_properties("${SRC}/core/arm64/fp_ops.cpp" PROPERTIES COMPILE_OPTIONS -frounding-math)
endif()
include(CTest)
if(BUILD_TESTING)
  file(GLOB ENGINE_TESTS CONFIGURE_DEPENDS "${NEXO}/tests/*.cpp")
  list(FILTER ENGINE_TESTS EXCLUDE REGEX "vulkan_tests[.]cpp$")
  add_executable(engine-tests ${ENGINE_TESTS})
  target_link_libraries(engine-tests PRIVATE switchcore)
  target_compile_definitions(engine-tests PRIVATE NEXO2_TEST_DATA_DIR="${NEXO}/tests/generated")
  add_test(NAME engine COMMAND engine-tests --interp)
  set_tests_properties(engine PROPERTIES TIMEOUT 240)
endif()

target_sources(switchcore PRIVATE core/runtime/SwitchCore.cpp)
if(BUILD_TESTING)
  add_executable(scanout-bench tools/scanout-bench.cpp)
  target_link_libraries(scanout-bench PRIVATE switchcore)
  add_executable(switchcore-replay tools/replay.cpp)
  target_link_libraries(switchcore-replay PRIVATE switchcore)
  add_executable(switchcore-probe tools/probe.cpp)
  add_custom_command(TARGET switchcore-probe POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CMAKE_CURRENT_SOURCE_DIR}/apps/ios/Resources/AtkinsonHyperlegible-Regular.ttf"
      "$<TARGET_FILE_DIR:switchcore-probe>/AtkinsonHyperlegible-Regular.ttf")
  target_link_libraries(switchcore-probe PRIVATE switchcore)
  add_executable(runtime-tests tests/integration/runtime_tests.cpp)
  target_link_libraries(runtime-tests PRIVATE switchcore)
  add_test(NAME runtime-contract COMMAND runtime-tests "${CMAKE_CURRENT_SOURCE_DIR}/apps/ios/Resources/rally.nro" "${CMAKE_CURRENT_BINARY_DIR}/contract-title")
  add_test(NAME interactive-rally COMMAND switchcore-probe "${CMAKE_CURRENT_SOURCE_DIR}/apps/ios/Resources/rally.nro" "${CMAKE_CURRENT_BINARY_DIR}/rally-title" rally "${CMAKE_CURRENT_BINARY_DIR}/rally.ppm")
endif()

if(BUILD_TESTING AND APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
  add_executable(native-probes tools/native-probes.cpp core/native/Probes.cpp)
  target_include_directories(native-probes PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME native-probes COMMAND native-probes)
endif()

# ARM64 recompiler (docs/nativo.md, milestone M1), on Apple arm64 hosts for now. The Windows build
# and the iOS project stay as they are.
if(APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
  set(SWITCHCORE_JIT_DEFAULT ON)
else()
  set(SWITCHCORE_JIT_DEFAULT OFF)
endif()
option(SWITCHCORE_JIT "Dynarmic recompiler for the guest CPU" ${SWITCHCORE_JIT_DEFAULT})
if(SWITCHCORE_JIT)
  include(cmake/Dynarmic.cmake)
  target_sources(switchcore PRIVATE core/runtime/CodePool.cpp)
  if(BUILD_TESTING)
    add_test(NAME engine-jit COMMAND engine-tests --jit)
    set_tests_properties(engine-jit PROPERTIES TIMEOUT 240)
    add_test(NAME interactive-rally-jit COMMAND switchcore-probe --jit "${CMAKE_CURRENT_SOURCE_DIR}/apps/ios/Resources/rally.nro" "${CMAKE_CURRENT_BINARY_DIR}/rally-jit-title" rally "${CMAKE_CURRENT_BINARY_DIR}/rally-jit.ppm")
    add_executable(jit-bench tests/integration/jit_bench.cpp)
    target_link_libraries(jit-bench PRIVATE switchcore)
    add_test(NAME jit-bench COMMAND jit-bench)
  endif()
endif()
